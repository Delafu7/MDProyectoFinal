\begin{alertblock}{Objetivos}
\begin{itemize}
    \item Evaluar cómo generalizan los modelos entre trastornos (cross-condition).
    \item Comparar dos representaciones lingüísticas:
    \begin{itemize}
        \item TF-IDF (frecuencias)
        \item RoBERTuito (embeddings contextualizados)
    \end{itemize}
    \item Comparar clasificadores:
    \begin{itemize}
        \item Dummy (baseline)
        \item MLP
    \end{itemize}
\item Research Questions (RQs):
\begin{itemize}
    \item \textbf{RQ1:} ¿Cuál es la mejor representación para el modelado clásico en tareas cross-condition — TF-IDF o RoBERTuito embeddings?
    \item \textbf{RQ2:} ¿Qué clasificador generaliza mejor entre condiciones?
\end{itemize}

\end{itemize}

\end{alertblock}



%----------------------------------------------------------------------------------------
%	QUICK REVISION

\begin{block}{Task \& Data}
\begin{itemize}
    \item \textbf{Tarea:} predecir \textit{sufre / no sufre}.
    \item \textbf{Cross-condition:} entrenar en (A) y evaluar en (B):
    \begin{itemize}
        \item Depresión → Ansiedad / TCA
        \item Ansiedad → Depresión
        \item TCA → Depresión
    \end{itemize}
    \item \textbf{Datos:} mensajes reales anonimizados por condición.
\end{itemize}
\end{block}

% Describe the data before and after the data cleaning in a tabular way
% Data characterization: describe the features used to represent each text
\begin{block}{Representaciones del texto}
\begin{itemize}
    \item \textbf{TF-IDF}: n-gramas (1–2), 5000 features, stopwords en español.
    \item \textbf{RoBERTuito}:
    \begin{itemize}
        \item Embeddings SentenceTransformer.
        \item Probabilidades de sentimiento del modelo RoBERTuito-SA.
        \item Distancia coseno a un mensaje clínicamente representativo.
        \item $\Rightarrow$ Representación final = [Embeddings + Sentiment + Distancia].
    \end{itemize}
\end{itemize}
\end{block}

\textbf{Discussion}
\begin{itemize}
    \item RoBERTuito generaliza mejor entre condiciones clínicas que TF-IDF.
    \item TF-IDF sigue funcionando bien con modelos lineales, aunque es menos estable.
    \item Los errores aparecen sobre todo en textos breves o ambiguos.
    \item Ejemplo (parafraseado): “No duermo bien y me siento raro” → difícil distinguir entre ansiedad/depresión.
\end{itemize}

\begin{alertblock}{Conclusiones}
\begin{itemize}
    \item RoBERTuito ofrece el mejor rendimiento global.
    \item TF-IDF es competitivo en algunos casos, pero menos consistente.
    \item MLP, SVM y RF superan claramente al Dummy. 
    \item Las transferencias entre condiciones se acerca a la simetría.
\end{itemize}
